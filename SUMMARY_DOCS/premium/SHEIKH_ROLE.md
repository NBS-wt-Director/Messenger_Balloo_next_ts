-- Таблица пользователей
CREATE TABLE users (
  id VARCHAR(36) PRIMARY KEY,

  display_name VARCHAR(100) NOT NULL,

  -- Статистика
  sheikh_score INT DEFAULT 0,
  referral_count INT DEFAULT 0,
  
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW() ON UPDATE NOW()
);

-- Таблица подписок
CREATE TABLE subscriptions (
  id VARCHAR(36) PRIMARY KEY,
  user_id VARCHAR(36) NOT NULL,
  plan_type ENUM('monthly', 'yearly', 'lifetime') NOT NULL,
  status ENUM('active', 'cancelled', 'expired') NOT NULL,
  
  amount DECIMAL(10, 2) NOT NULL,
  currency VARCHAR(3) DEFAULT 'USD',
  
  started_at TIMESTAMP NOT NULL,
  expires_at TIMESTAMP NOT NULL,
  cancelled_at TIMESTAMP NULL,
  
  payment_method VARCHAR(50),
  transaction_id VARCHAR(100),
  
  FOREIGN KEY (user_id) REFERENCES users(id),
  INDEX idx_user_id (user_id),
  INDEX idx_expires_at (expires_at)
);

-- Таблица транзакций
CREATE TABLE transactions (
  id VARCHAR(36) PRIMARY KEY,
  user_id VARCHAR(36) NOT NULL,
  type ENUM('subscription', 'payment', 'refund') NOT NULL,
  amount DECIMAL(10, 2) NOT NULL,
  currency VARCHAR(3) DEFAULT 'USD',
  status ENUM('pending', 'completed', 'failed', 'refunded') NOT NULL,
  
  description TEXT,
  metadata JSON,
  
  created_at TIMESTAMP DEFAULT NOW(),
  
  FOREIGN KEY (user_id) REFERENCES users(id),
  INDEX idx_user_id (user_id),
  INDEX idx_created_at (created_at)
);

## 🔐 ПРОВЕРКА ДОСТУПА

### Middleware:

```typescript
// middleware/sheikh.ts
import { Request, Response, NextFunction } from 'express';
import { PREMIUM_ATTACHMENTS, AttachmentType } from '@balloo/core-types';

export interface SheikhRequest extends Request {
  isSheikh?: boolean;
  sheikhType?: string;
}

export async function sheikhMiddleware(
  req: SheikhRequest,
  res: Response,
  next: NextFunction
): Promise<void> {
  
  const userId = req.user?.id;
  if (!userId) {
    return res.status(401).json({
      success: false,
      error: { code: 'UNAUTHORIZED', message: 'Требуется авторизация' }
    });
  }
  
  const user = await getUserById(userId);
  
  req.isSheikh = user.isSheikh;
  req.sheikhType = user.sheikhType;
  
  next();
}

export function requireSheikh(attachmentType?: AttachmentType) {
  return async (req: SheikhRequest, res: Response, next: NextFunction) => {
    
    const type = attachmentType || req.params.type;
    
    // Если вложение premium и пользователь не Шейх
    if (PREMIUM_ATTACHMENTS.includes(type) && !req.isSheikh) {
      return res.status(403).json({
        success: false,
        error: {
          code: 'PREMIUM_REQUIRED',
          message: 'Требуется статус Шейх',
          upgradeUrl: '/api/v1/premium/upgrade',
          pricing: {
            monthly: '$9.99/мес',
            yearly: '$99.99/год',
            lifetime: '$299.99'
          },
          benefits: [
            '5 premium вложений',
            'Файлы до 500 MB',
            '0% комиссия на переводы',
            'Приоритетная поддержка 24/7'
          ]
        }
      });
    }
    
    // Проверка срока действия
    if (req.isSheikh && req.user?.sheikhExpiresAt) {
      if (req.user.sheikhExpiresAt < Date.now()) {
        return res.status(403).json({
          success: false,
          error: {
            code: 'PREMIUM_EXPIRED',
            message: 'Статус Шейх истёк',
            renewUrl: '/api/v1/premium/renew',
            discount: 0.20
          }
        });
      }
    }
    
    next();
  };
}
```
