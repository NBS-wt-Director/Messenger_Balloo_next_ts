✅ Technical UI Invariants

Critical design rules for all technical nodes

Must Follow Rules
  • border-radius: 0 — No rounded corners anywhere
  • border: 2px solid — All borders 2px
  • Status always visible — Health/state in header
  • Keyboard shortcuts — All actions accessible via keyboard
  • Logs accessible — Log viewer available
  • Auth required — No public technical UI
  • Audit trail — Actions logged and visible
// Technical Design Context
interface TechnicalDesignContext {
  nodeType: 'technical';
  requiresAuth: true;
  showHealth: true;
  showLogs: true;
  allowRollback: true;
  borderRadius: 0; // Always
  borderWidth: 2; // Default
}
